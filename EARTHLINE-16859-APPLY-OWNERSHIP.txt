function applyOwnership(mode){
    if(!mapReady())return false;
    const property=mode==="property";
    const validationView=property?String(state.validationView15806||"all"):"all";
    swalesOnly=validationView==="swales-only";
    const swalesContours=validationView==="swales-contours";
    const active=registry[property?"property":"regional"];
    const inactive=registry[property?"regional":"property"];
    setGroup(inactive.water,false);setGroup(inactive.swales,false);
    if(property)for(const id of obsoletePropertyLayers)visible(id,false);
    const propertyCommitted=!!(
      !property||
      (
        M&&M.propertyPublication15816&&
        M.propertyPublication15816.committed===true&&
        Number(M.propertyPublication15816.rawGeographicSegments||0)>0
      )
    );
    const flowOn=propertyCommitted&&(
      property
        ?true
        :!!(M.layers&&M.layers.flow&&M.layers.flow.on)
    )&&validationView==="all";
    const swaleOn=propertyCommitted&&!!(M.layers&&M.layers.swales&&M.layers.swales.on);
    const rechargeOn=propertyCommitted&&!!(M.layers&&M.layers.recharge&&M.layers.recharge.on)&&validationView==="all";
    const contourOn=propertyCommitted&&!!(M.layers&&M.layers.contours&&M.layers.contours.on)&&(validationView==="all"||swalesContours);
    const aquiferOn=propertyCommitted&&!state.propertyAquiferHold16359&&!!(M.layers&&M.layers.usgsAquifer&&M.layers.usgsAquifer.on)&&validationView==="all";
    const aquiferTextOn=propertyCommitted&&!state.propertyAquiferHold16359&&validationView==="all"&&!!(
      (M&&M.vtGroundwaterSummaryPoint16398)||
      (M&&Array.isArray(M.usgsAquifers)&&M.usgsAquifers.length)
    );
    setGroup(active.water,flowOn);setGroup(active.swales,swaleOn);
    setGroup(active.recharge,rechargeOn);setGroup(active.contours,contourOn);setGroup(active.aquifer,aquiferOn);
    updateAquiferTopText16501(property,aquiferTextOn);
    /* EARTHLINE 16414 — authoritative hydro evidence visibility order.
       Keep the translucent recharge fill contextual below interventions, but guarantee
       that an actually selected/published recharge footprint cannot disappear beneath
       the later living-swale/planting renderer. Annotation visibility remains owned here. */
    try{
      if(earthlineMap.getLayer("earthline-recharge-label")){
        earthlineMap.setLayoutProperty("earthline-recharge-label","text-allow-overlap",false);
        earthlineMap.setLayoutProperty("earthline-recharge-label","text-ignore-placement",false);
      }
      for(const id16414 of [
        "earthline-recharge-line",
        "earthline-recharge-highlight",
        "earthline-recharge-label",
        "earthline-aquifer-wells-16398"
      ]){
        if(earthlineMap.getLayer(id16414))earthlineMap.moveLayer(id16414);
      }
      /* 16492: all aquifer language is the final annotation pass. Geometry stays
         in its governed order; only text is moved above every analytical layer. */
      for(const id16492 of [
        "el-live-aquifer-label-15970",
        "earthline-aquifer-label-points-layer-15788",
        "earthline-aquifer-evidence-label-16398"
      ]){
        if(earthlineMap.getLayer(id16492))earthlineMap.moveLayer(id16492);
      }
    }catch(_){ }
    try{
      const rechargeCanvas=document.getElementById("earthlineRechargeCanvas");
      const rechargeSvg=document.getElementById("earthlineRechargeOverlay");
      if(rechargeCanvas)rechargeCanvas.style.display="none";
      if(rechargeSvg){rechargeSvg.style.display="none";rechargeSvg.replaceChildren()}
    }catch(e){}
    const controls=ensureControl();
    if(controls){
      controls.button.classList.toggle("visible",property&&!!M.analysisReady);
      controls.badge.classList.toggle("visible",property&&swalesOnly&&!!M.analysisReady)
    }
    state.rendererOwnership15805={
      build:BUILD,mode:property?"property":"regional",swalesOnly,validationView,propertyCommitted,appliedAt:Date.now(),
      activeOwners:{water:active.water.owner,swales:active.swales.owner,recharge:active.recharge.owner,
        contours:active.contours.owner,aquifer:active.aquifer.owner}
    };
    return true
  }
