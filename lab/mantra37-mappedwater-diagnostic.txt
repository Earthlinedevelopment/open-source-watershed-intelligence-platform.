function earthlineDilateMask(src,r){
  const rr=Math.max(0,Math.ceil(r));
  if(!rr)return new Uint8Array(src);
  const out=new Uint8Array(N);
  for(let y=0;y<GH;y++)for(let x=0;x<GW;x++){
    if(!src[idx(x,y)])continue;
    for(let oy=-rr;oy<=rr;oy++)for(let ox=-rr;ox<=rr;ox++){
      if(ox*ox+oy*oy>rr*rr)continue;
      const nx=x+ox,ny=y+oy;
      if(nx>=0&&ny>=0&&nx<GW&&ny<GH)out[idx(nx,ny)]=1;
    }
  }
  return out;
}

function earthlineProcessMappedFeature(roadMask,buildingMask,f){
  if(!f||!f.geometry)return 0;
  const info=earthlineMappedFeatureClass15862J(f);
  if(!info.relevant)return 0;
  const g=f.geometry;let touched=0;

  if(g.type==="GeometryCollection"&&Array.isArray(g.geometries)){
    for(const geometry of g.geometries){
      touched+=earthlineProcessMappedFeature(
        roadMask,
        buildingMask,
        Object.assign({},f,{geometry})
      );
    }
    return touched;
  }

  if(g.type==="LineString"||g.type==="MultiLineString"){
    const lines=g.type==="LineString"?[g.coordinates]:g.coordinates;
    if(info.water){
      // EARTHLINE 16262 — mapped surface water is an absolute Property no-build input.
      const buf=/river|lake|reservoir/.test(info.cls)?22:14;
      for(const line of lines){
        earthlineRasterLine(roadMask,line,buf);
        touched++;
      }
    }else if(info.road){
      // Preserve the established physical buffers. This build expands feature
      // capture only; it does not widen any exclusion distance.
      const buf=/motorway|trunk|primary/.test(info.cls)?16:
        /secondary|tertiary/.test(info.cls)?13:
        /service|drive|driveway|parking_aisle|track|path|footway|cycleway|pedestrian/.test(info.cls)?10:11;
      for(const line of lines){
        earthlineRasterLine(roadMask,line,buf);
        touched++;
      }
    }else if(info.building){
      // Some styles expose building footprints only as outline layers.
      for(const line of lines){
        earthlineRasterLine(buildingMask,line,12);
        touched++;
      }
    }
  }else if(g.type==="Polygon"||g.type==="MultiPolygon"){
    const polys=g.type==="Polygon"?[g.coordinates]:g.coordinates;
    const target=info.building?buildingMask:roadMask;
    const bufferM=info.water?(/river|lake|reservoir/.test(info.cls)?22:14):(info.building?12:10);
    for(const rings of polys){
      earthlineRasterPolygon(target,rings,bufferM);
      touched++;
    }
  }
  return touched;
}
async function earthlineWaitForMappedExclusionStyle15862B(myGen=M.searchGen,timeoutMs=5000){
  if(!earthlineMapActive())return {ready:false,status:"map-inactive",waitedMs:0};
  const started=performance.now();
  /* EARTHLINE 16467 — performance only. Exclusion processing does not need the
     entire Mapbox style (imagery, glyphs, unrelated sources) to settle. It needs
     the exact road/building/water style layers used by vectorNoBuild15778 and the
     source(s) backing those layers to be queryable. Keep the same evidence set and
     fail-closed owner; narrow readiness to the resources this owner actually uses. */
  const styleHasLayers=()=>{
    try{
      if(!earthlineMap)return false;
      const s=earthlineMap.getStyle&&earthlineMap.getStyle();
      if(!s||!Array.isArray(s.layers)||!s.layers.length)return false;
      const relevant=s.layers.filter(l=>{
        const id=((l.id||"")+" "+(l["source-layer"]||"")).toLowerCase();
        const lineLike=l.type==="line";
        const areaLike=l.type==="fill"||l.type==="fill-extrusion";
        const transport=/road|street|bridge|tunnel|path|track|service|transportation|highway/.test(id);
        const structure=/building|structure|parking/.test(id);
        const mappedWater=/water|river|lake|reservoir|stream|canal|wetland/.test(id);
        const genericParkingCarrier=areaLike&&/landuse|land-use/.test(id);
        return (lineLike&&(transport||structure||mappedWater))||
               (areaLike&&(transport||structure||genericParkingCarrier||mappedWater));
      });
      if(!relevant.length)return false;
      if(typeof earthlineMap.isSourceLoaded==="function"){
        const sources=[...new Set(relevant.map(l=>l.source).filter(Boolean))];
        for(const sourceId of sources){
          try{if(!earthlineMap.isSourceLoaded(sourceId))return false;}catch(_){return false;}
        }
      }
      return true;
    }catch(e){return false;}
  };
  if(styleHasLayers()){
    try{await new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r)));}catch(e){}
    return {ready:true,status:"already-ready",waitedMs:Math.round(performance.now()-started)};
  }
  return await new Promise(resolve=>{
    let done=false;
    const finish=(ready,status)=>{
      if(done)return;done=true;
      clearInterval(poll);clearTimeout(limit);
      try{
        earthlineMap.off("styledata",check);
        earthlineMap.off("sourcedata",check);
        earthlineMap.off("load",check);
        earthlineMap.off("idle",check);
      }catch(e){}
      resolve({ready,status,waitedMs:Math.round(performance.now()-started)});
    };
    const check=()=>{
      if(typeof earthlineIsStale==="function"&&earthlineIsStale(myGen))return finish(false,"stale");
      if(styleHasLayers()){
        // Two animation frames allow the freshly loaded style to become queryable.
        try{
          requestAnimationFrame(()=>requestAnimationFrame(()=>finish(true,"style-ready")));
        }catch(e){finish(true,"style-ready");}
      }
    };
    const poll=setInterval(check,100);
    const limit=setTimeout(()=>finish(false,"timeout"),Math.max(500,timeoutMs));
    try{
      earthlineMap.on("styledata",check);
      earthlineMap.on("sourcedata",check);
      earthlineMap.on("load",check);
      earthlineMap.on("idle",check);
    }catch(e){}
    check();
  });
}
async function earthlineBuildVectorNoBuildMask(myGen=M.searchGen){
  let mask=new Uint8Array(N),roadMask=new Uint8Array(N),buildingMask=new Uint8Array(N),count=0;
  M.vectorNoBuildCoverage={
    styleReady:false,layerCount:0,featureCount:0,processedFeatureCount:0,verified:false,
    owner:"earthlineBuildVectorNoBuildMask",build:"BUILD-CHECK-15862B",
    readinessWait:{status:"not-started",waitedMs:0}
  };
  const readiness=await earthlineWaitForMappedExclusionStyle15862B(myGen,5000);
  M.vectorNoBuildCoverage.readinessWait=readiness;
  if(!readiness.ready){
    M.vectorNoBuildMask=mask;M.vectorNoBuildCount=0;
    M.vectorNoBuildCoverage.unverifiedReason="map-style-not-queryable";
    return false;
  }
  M.vectorNoBuildCoverage.styleReady=true;
  try{
    if(earthlineIsStale(myGen))return false;
    const style=earthlineMap.getStyle()||{};
    const layerIds=(style.layers||[]).filter(l=>{
      const id=(l.id||"").toLowerCase();
      return (l.type==="line"&&/road|street|bridge|tunnel|path|track|service/.test(id))||
             (l.type==="fill"&&/building|structure|parking/.test(id));
    }).map(l=>l.id);
    M.vectorNoBuildCoverage.layerCount=layerIds.length;
    M.vectorNoBuildCoverage.verified=layerIds.length>0;
    const c=earthlineMap.getCanvas();
    let queryBox=[[0,0],[c.clientWidth,c.clientHeight]];
    if(Array.isArray(M.analysisBoundaryLngLat)&&M.analysisBoundaryLngLat.length>=3){
      const pts=M.analysisBoundaryLngLat.map(p=>earthlineMap.project(p));
      const pad=36;
      const minX=clamp(Math.min(...pts.map(p=>p.x))-pad,0,c.clientWidth);
      const maxX=clamp(Math.max(...pts.map(p=>p.x))+pad,0,c.clientWidth);
      const minY=clamp(Math.min(...pts.map(p=>p.y))-pad,0,c.clientHeight);
      const maxY=clamp(Math.max(...pts.map(p=>p.y))+pad,0,c.clientHeight);
      queryBox=[[minX,minY],[maxX,maxY]];
    }
    let features=layerIds.length?(earthlineMap.queryRenderedFeatures(queryBox,{layers:layerIds})||[]):[];
    // Rendered features are sufficient for a fast first-pass exclusion mask. Querying every
    // loaded source layer returned thousands of off-parcel features and could freeze analysis.
    const seen=new Set(),bounded=[];
    for(const f of features){
      const key=(f.id!=null?f.id:"")+"|"+(f.sourceLayer||"")+"|"+((f.layer&&f.layer.id)||"");
      if(seen.has(key))continue;seen.add(key);bounded.push(f);
      if(bounded.length>=320)break;
    }
    M.vectorNoBuildCoverage.featureCount=bounded.length;
    for(let n=0;n<bounded.length;n++){
      if(earthlineIsStale(myGen))return false;
      count+=earthlineProcessMappedFeature(roadMask,buildingMask,bounded[n]);
      if((n&15)===15){earthlineProgress("hydrology","mapping roads and structures…");await earthlineYield();}
    }
    const seamM=2.5;
    roadMask=earthlineDilateMask(roadMask,Math.max(1,seamM/Math.max(0.2,M.cellM||1)));
    for(let i=0;i<N;i++)mask[i]=(roadMask[i]||buildingMask[i])?1:0;
  }catch(err){console.warn("[Earthline] mapped road/building mask unavailable",err);}
  M.vectorNoBuildMask=mask;M.vectorNoBuildCount=count;
  M.vectorNoBuildStamp=[M.centerLat.toFixed(6),M.centerLng.toFixed(6),M.viewZoom,count].join("|");
  M._noBuildStamp="";
  if(M.vectorNoBuildCoverage){
    M.vectorNoBuildCoverage.processedFeatureCount=count;
    M.vectorNoBuildCoverage.renderedFeatureCount=Number(M.vectorNoBuildCoverage.featureCount||0);
    M.vectorNoBuildCoverage.featureCount=count;
    M.vectorNoBuildCoverage.verified=!!(
      M.vectorNoBuildCoverage.styleReady&&
      M.vectorNoBuildCoverage.layerCount>0&&
      count>0
    );
    if(!M.vectorNoBuildCoverage.verified){
      M.vectorNoBuildCoverage.unverifiedReason=count>0?null:
        (M.vectorNoBuildCoverage.layerCount>0?"no-road-or-building-features-in-analysis-frame":"no-relevant-style-layers");
    }
  }
  console.log("[Earthline] mapped exclusion features:",count,M.vectorNoBuildCoverage);
  return !!(M.vectorNoBuildCoverage&&M.vectorNoBuildCoverage.verified);
}
