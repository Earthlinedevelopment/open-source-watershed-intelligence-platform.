  // For large areas (states, countries, regions) Mapbox returns a bbox [W,S,E,N].
  // Derive a zoom that fits the whole area to the map window so e.g. "Colorado" frames
  // the entire state rather than centering mid-state at a fixed zoom.
  let fitZoom=zh, bbox=f.bbox||null;

    // so solve 156573.03*cos/2^(z+1) = targetMpp  ->  z = log2(156573*cos/(2*targetMpp)).
    const targetMpp=(spanKm*1000*2.10)/(GW*5);   // 14300: full state plus surrounding-state edges, faster state framing
    const c=Math.cos(f.center[1]*Math.PI/180);
    fitZoom=clamp(Math.log2(156573.03*c/(2*Math.max(targetMpp,0.01))),3,18);
  }
  const countryFeature=((f.place_type||[]).includes("country")?f:(f.context||[]).find(c=>String(c.id||"").startsWith("country.")))||null;

function earthlineIsPropertyLocation(loc){
  if(!loc)return false;
  if(loc.planning===true)return false;
  if(loc.userSelectedSite===true)return true;
  if(GAZ.includes(loc))return true;
  if(Array.isArray(loc.zones)&&loc.zones[0]==="Full Parcel")return true;
  const pt=(loc.placeType||"").toLowerCase();
  if(pt==="address"||pt==="poi"||pt==="neighborhood")return true;
  const q=((loc.q||"")+" "+(loc.name||"")+" "+(loc.fullName||"")+" "+(loc.sub||"")).toLowerCase();
  if(/(^|\b)(address|parcel|property|farm|homestead|orchard|ranch|smallholding)(\b|$)/.test(q))return true;
  // Numeric street addresses and common road suffixes must be treated as property-scale
  // even when Mapbox labels the result as a place/locality rather than an address.
  if(/\b\d{1,6}\s+[a-z0-9.'-]+(?:\s+[a-z0-9.'-]+){0,5}\s+(rd|road|st|street|ave|avenue|dr|drive|ln|lane|ct|court|way|hwy|highway|trl|trail|pike|route)\b/i.test(q))return true;
  // Small non-regional results without a large bbox are safe to promote to working zoom.
  if(!loc.bbox && Number(loc.zoomHint||0)>=14)return true;
  return false;
}
